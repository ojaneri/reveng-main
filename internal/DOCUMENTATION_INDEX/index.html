    Essential Commands¶

    # Installation
./install-reveng.sh

# First analysis
reveng analyze /path/to/binary

# JavaScript deobfuscation
./reveng-js deobfuscate file.js

# MCP server
./reveng-mcp-server

# Get help
reveng --help
./reveng-js --help
./reveng-mcp-server --help
